# SQL Interview Practice Dataset

## Schema

### users

```sql
CREATE TABLE users (
    user_id INT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(255),
    country VARCHAR(50),
    signup_date DATE
);
```

### products

```sql
CREATE TABLE products (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(255),
    category VARCHAR(100),
    price DECIMAL(10,2)
);
```

### orders

```sql
CREATE TABLE orders (
    order_id INT PRIMARY KEY,
    user_id INT,
    order_date DATE,
    status VARCHAR(50),
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);
```

### order_items

```sql
CREATE TABLE order_items (
    order_item_id INT PRIMARY KEY,
    order_id INT,
    product_id INT,
    quantity INT,
    unit_price DECIMAL(10,2),
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (product_id) REFERENCES products(product_id)
);
```

### payments

```sql
CREATE TABLE payments (
    payment_id INT PRIMARY KEY,
    order_id INT,
    payment_date DATE,
    amount DECIMAL(10,2),
    payment_method VARCHAR(50),
    FOREIGN KEY (order_id) REFERENCES orders(order_id)
);
```

### employees

```sql
CREATE TABLE employees (
    employee_id INT PRIMARY KEY,
    employee_name VARCHAR(100),
    manager_id INT,
    department VARCHAR(50),
    FOREIGN KEY (manager_id) REFERENCES employees(employee_id)
);
```

# Solutions

## Q1. Users from India

```sql
SELECT *
FROM users
WHERE country = 'India';
```

---

## Q2. Orders placed in 2025

```sql
SELECT *
FROM orders
WHERE order_date >= '2025-01-01'
  AND order_date < '2026-01-01';
```

## Q3. Order details with product names

```sql
SELECT
    o.order_id,
    p.product_name,
    oi.quantity
FROM orders o
JOIN order_items oi
    ON o.order_id = oi.order_id
JOIN products p
    ON oi.product_id = p.product_id;
```

## Q4. Users who never placed an order

## Q6. Revenue generated by each product


## Q8. Average order value per user

```sql
SELECT
    u.user_id,
    u.name,
    AVG(order_total.order_value) AS avg_order_value
FROM users u
JOIN (
    SELECT
        order_id,
        SUM(quantity * unit_price) AS order_value
    FROM order_items
    GROUP BY order_id
) order_total
JOIN orders o
    ON order_total.order_id = o.order_id
    AND u.user_id = o.user_id
GROUP BY u.user_id, u.name;
```

## Q16. Monthly spending and running total

```sql
WITH monthly AS (
    SELECT
        o.user_id,
        DATE_TRUNC('month', o.order_date) AS month,
        SUM(pay.amount) AS monthly_spending
    FROM orders o
    JOIN payments pay
        ON o.order_id = pay.order_id
    GROUP BY o.user_id, DATE_TRUNC('month', o.order_date)
)
SELECT
    *,
    SUM(monthly_spending)
        OVER (
            PARTITION BY user_id
            ORDER BY month
        ) AS running_total
FROM monthly;
```

## Q21. Users whose spending increased every month

```sql
WITH monthly AS (
    SELECT
        user_id,
        DATE_TRUNC('month', order_date) AS month,
        SUM(amount) AS spending
    FROM orders o
    JOIN payments p
        ON o.order_id = p.order_id
    GROUP BY user_id, DATE_TRUNC('month', order_date)
),
compare AS (
    SELECT *,
           LAG(spending)
           OVER (
               PARTITION BY user_id
               ORDER BY month
           ) prev_spending
    FROM monthly
)
SELECT user_id
FROM compare
GROUP BY user_id
HAVING BOOL_AND(
    prev_spending IS NULL
    OR spending > prev_spending
);
```

## Q23. Customers ordering in 3 consecutive months

```sql
WITH months AS (
    SELECT DISTINCT
        user_id,
        DATE_TRUNC('month', order_date) AS month
    FROM orders
),
numbered AS (
    SELECT *,
           ROW_NUMBER()
           OVER (
               PARTITION BY user_id
               ORDER BY month
           ) rn
    FROM months
)
SELECT user_id
FROM numbered
GROUP BY user_id
HAVING COUNT(*) >= 3;
```

## Q25. Employees reporting indirectly to CEO

```sql
WITH RECURSIVE hierarchy AS (
    SELECT
        employee_id,
        employee_name,
        manager_id,
        0 AS depth
    FROM employees
    WHERE employee_name = 'CEO'

    UNION ALL

    SELECT
        e.employee_id,
        e.employee_name,
        e.manager_id,
        h.depth + 1
    FROM employees e
    JOIN hierarchy h
        ON e.manager_id = h.employee_id
)
SELECT *
FROM hierarchy
WHERE depth > 1;
```

## Q27. Revenue contribution percentage

```sql
WITH revenue AS (

        SUM(oi.quantity * oi.unit_price) AS revenue
    FROM products p
    JOIN order_items oi
        ON p.product_id = oi.product_id
    GROUP BY p.product_name
)
SELECT
    product_name,
    revenue,
    ROUND(
        revenue * 100.0 /
```

## Q29. Users responsible for top 80% revenue

```sql
WITH spending AS (
    SELECT
        u.user_id,
        u.name,
        SUM(pay.amount) AS total_spent
    FROM users u
    JOIN orders o
        ON u.user_id = o.user_id
    JOIN payments pay
        ON o.order_id = pay.order_id
    GROUP BY u.user_id, u.name
),
ranked AS (
    SELECT *,
           SUM(total_spent)
           OVER (
               ORDER BY total_spent DESC
           ) running_total,
           SUM(total_spent)
           OVER () grand_total
    FROM spending
)
SELECT *
FROM ranked
WHERE running_total <= grand_total * 0.8;
```

## Q30. Frequently purchased together

```sql
SELECT
    p1.product_name AS product_a,
    p2.product_name AS product_b,
    COUNT(*) AS times_bought_together
FROM order_items oi1
JOIN order_items oi2
    ON oi1.order_id = oi2.order_id
   AND oi1.product_id < oi2.product_id
JOIN products p1
    ON oi1.product_id = p1.product_id
JOIN products p2
    ON oi2.product_id = p2.product_id
GROUP BY
    p1.product_name,
    p2.product_name
ORDER BY times_bought_together DESC;
```
